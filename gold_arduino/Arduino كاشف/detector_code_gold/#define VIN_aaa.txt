#define VIN_PIN A5
#define BUZZER_PIN 9
#define PULSE_PIN A4
#define LED_PIN 10
#define RESET_BTN_PIN 2
#define TRIGGER_BTN_PIN 3
#define SENSITIVITY_POT_APIN A1
#define SPEAKER_PIN 5
#define BASE_TONE_FREQUENCY 1000
#define SPINNER_THRESHOLD 10
#define SPINNER_PIN 6
#define METAL_TYPE_1_THRESHOLD 100
#define METAL_TYPE_2_THRESHOLD 200

long sumExpect = 0;
long ignor = 0;
long diff = 0;
long pTime = 0;
long buzPeriod = 0;
unsigned long metalDetectedTime = 0;
unsigned long metalDetectDuration = 500;
int threshold = 500;
int storedTimeDeltaDifference = 0;

void setup() {
  Serial.begin(9600);
  pinMode(PULSE_PIN, OUTPUT);
  digitalWrite(PULSE_PIN, LOW);
  pinMode(VIN_PIN, INPUT);
  pinMode(BUZZER_PIN, OUTPUT);
  digitalWrite(BUZZER_PIN, LOW);
  pinMode(LED_PIN, OUTPUT);
  pinMode(RESET_BTN_PIN, INPUT_PULLUP);
  pinMode(TRIGGER_BTN_PIN, INPUT_PULLUP);
  pinMode(SPINNER_PIN, OUTPUT);
}

void loop() {
  long int diff = 0;
  char buzState = 0;

  // التميز بين المعادن
  if (digitalRead(TRIGGER_BTN_PIN) == LOW) {
    // tone(SPEAKER_PIN, BASE_TONE_FREQUENCY + storedTimeDeltaDifference);

    if (storedTimeDeltaDifference > SPINNER_THRESHOLD) {
      digitalWrite(SPINNER_PIN, HIGH);
      // ... الشيفرة المتعلقة بالكشف عن المعدن
    } else {
      digitalWrite(SPINNER_PIN, LOW);
    }
  } else {
    noTone(SPEAKER_PIN);
    digitalWrite(SPINNER_PIN, LOW);
  }

  if (digitalRead(RESET_BTN_PIN) == LOW) {
    storedTimeDeltaDifference = 0;
  }

  int minVal = 1023;
  int maxVal = 0;
  long unsigned int sum = 0;

  for (int i = 0; i < 256; i++) {
    // reset the capacitor
    pinMode(VIN_PIN, OUTPUT);
    digitalWrite(VIN_PIN, LOW);
    delayMicroseconds(20);
    pinMode(VIN_PIN, INPUT);
    applyPulses();

    // read the charge of capacitor
    int val = analogRead(VIN_PIN);
    minVal = min(val, minVal);
    maxVal = max(val, maxVal);
    sum += val;

    manageBuzzer(buzState);
  }

  sum -= minVal;
  sum -= maxVal;

  // إذا تم اكتشاف المعدن
  if (diff != 0 && abs(diff) >= threshold) {
    // إعادة تعيين وقت اكتشاف المعدن
    metalDetectedTime = millis();

    // قم بتشغيل المنبه والإشارة LED
    if (buzState == 1) {
      tone(BUZZER_PIN, 2000);
      digitalWrite(LED_PIN, HIGH);
    } else if (buzState == 2) {
      tone(BUZZER_PIN, 500);
      digitalWrite(LED_PIN, HIGH);
    }
  }

  // إذا مرت فترة الاستمرار المحددة بعد اكتشاف المعدن، قم بإيقاف الصوت
  if (millis() - metalDetectedTime >= metalDetectDuration) {
    noTone(BUZZER_PIN);
    digitalWrite(LED_PIN, LOW);
  }

  // subtract minimum and maximum value to remove spikes
  sum -= minVal;
  sum -= maxVal;

  if (sumExpect == 0) 
    sumExpect = sum << 6; // set sumExpect to expected value
  long int avgsum = (sumExpect - 32) >> 6; 
  diff = sum - avgsum;

  // ... (بقية الشيفرة)
}

void applyPulses() {
  for (int i = 0; i < 3; i++) {
    digitalWrite(PULSE_PIN, HIGH); // take 3.5 uS
    delayMicroseconds(2);
    digitalWrite(PULSE_PIN, LOW);  // take 3.5 uS
    delayMicroseconds(2);
  }
}

void manageBuzzer(char buzStateParam) {
  long unsigned int currentTime = millis();

  if (currentTime < pTime + 10 || currentTime > pTime + buzPeriod) {
    if (diff > 0)
      buzStateParam = 1;
    else if (diff < 0)
      buzStateParam = 2;

    pTime = currentTime;
  }

  if (buzPeriod > 300)
    buzStateParam = 0;

  if (buzStateParam == 0) {
    digitalWrite(LED_PIN, LOW);
    noTone(BUZZER_PIN);
  } else if (buzStateParam == 1) {
     tone(BUZZER_PIN, 2000);
    digitalWrite(LED_PIN, HIGH);
  } else if (buzStateParam == 2) {
    tone(BUZZER_PIN, 500);
    digitalWrite(LED_PIN, HIGH);
  }
}
